# Tray double-click feature test.
#
# Double-clicking the tray icon opens the input screen right away, skipping
# the countdown. Runs headless in one process (app_test.exe): `app init`
# creates the real (hidden) windows without a message loop, `app tick`
# simulates pending timers, and `app tray-event` delivers the same OS events
# the real tray icon would (WM_LBUTTONDBLCLK).
#
# Covers the per-feature checklist:
#   1. happy path    - double-click -> input visible, countdown canceled
#   2. complete      - type the answer -> info screen, no stale timers
#   3. repeat        - double-click again (while info is showing) -> input
#                       reopens with the answer cleared
#   3b. real sequence - a real double-click delivers UP, DBLCLK, UP; the
#                       trailing UP must not leave the menu open
#   4. side effects  - no pending timers, only the expected windows
#   5. interleaving  - stop right after arming the input-done timer; a later
#                       tick must not bring the info window back
#   6. while stopped - double-click reopens input even when the app is paused

# --- setup: app is running, countdown going (shared preamble) ---
include ../common/running.txt

# --- 1. happy path: double-click -> input opens right away ---
app tray-event double-click
app assert visible input == true
app assert visible countdown == false
window input
assert value #answer == ""
app assert inputDone == false
# the in-flight countdown must be canceled (no phantom ticks)
app assert timers == 0

# --- 2. complete the word -> info screen ---
type #answer ringo
app assert inputDone == true
app tick
app assert visible info == true
app assert visible input == false
app assert timers == 0

# --- 3. repeat: double-click again (while info is showing) -> input reopens
# with the answer cleared. NOTE: the word is NOT re-picked here - only
# start_countdown() and tray-open pick a new word; show_input() just clears
# the value, which is what the assert below checks. ---
app tray-event double-click
app assert visible input == true
app assert visible info == false
window input
assert value #answer == ""

# --- 3b. real tray sequence: a real double-click delivers UP, DBLCLK, UP.
# Left clicks no longer open the tray menu (only right-click does), so the
# menu must never appear anywhere in this sequence. ---
app tray-event left-click
app assert visible tray == false
app assert visible input == true
app tray-event double-click
app assert visible tray == false
app assert visible input == true
app tray-event left-click
app assert visible tray == false
app assert visible input == true

# --- 3c. right-click is the only way to open the tray menu ---
app tray-event right-click
app assert visible tray == true

window tray
click #tray-open

# --- 4. side effects: no stale timers, only expected windows. tray-open
# re-picks a word and rebuilds the word windows, so the old info host is
# parked in the graveyard (not left alive). ---
app assert timers == 0
app assert inputWin == true
app assert infoWin == false
app assert stopped == false

# --- 5. interleaving: arm the input-done timer, then stop; a later tick
# must not bring the info window back. tray-open re-picked a random word, so
# pin the fixture word again and drive the countdown to input. ---
window tray
click #tray-stop
click #tray-stop
app word src/tests/word.json
app tick 30
window input
type #answer ringo
app assert inputDone == true
app assert timers == 1
window tray
click #tray-stop
app assert stopped == true
app assert timers == 0
app tick
app assert visible info == false
app assert visible input == false
app assert timers == 0

# --- 6. while stopped: double-click reopens input (pause, not quit) ---
app tray-event double-click
app assert visible input == true
app assert inputDone == false
window input
assert value #answer == ""
